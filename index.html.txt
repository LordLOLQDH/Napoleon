        <h3 data-template-id="quiz-complete-title" class="canva-text mt-3 font-['Source_Serif_Pro'] font-bold" style="color: rgb(32, 57, 93); font-weight: 700; font-style: normal; font-size: 19px;">Quiz geschafft!</h3>
        <p id="quiz-complete-score" class="mt-2 text-lg font-bold text-[#315c92]"></p>
       </div>
       <div class="mt-6 flex items-center justify-between gap-4 border-t border-[#e8edf1] pt-5">
        <p id="save-status" class="text-sm text-[#52647a]" aria-live="polite"></p><button id="next-question" data-template-id="next-question" class="canva-button hidden rounded-full px-5 py-3 font-bold transition hover:-translate-y-0.5" type="button" style="background: rgb(32, 57, 93); color: rgb(255, 255, 255); font-weight: 700; font-style: normal; font-size: 16px;">Nächste Frage</button>
       </div>

   <footer class="w-full border-t border-[#d9cfb8] bg-[#f5f0e5]">
    <div class="mx-auto max-w-6xl px-5 py-6 text-center sm:px-8">
     <p data-template-id="footer-text" class="canva-text text-sm" style="color: rgb(82, 100, 122); font-weight: 400; font-style: normal; font-size: 14px;">Lernen heißt auch hinterfragen: Geschichte besteht aus unterschiedlichen Perspektiven und Folgen.</p>
    </div>
   </footer>

  <script>
    let backendRecords = [];
    let currentQuestion = 1;
    let localState = { quiz_score: 0, quiz_completed: false, cards_viewed: 0, timeline_viewed: 0 };
    const viewedYears = new Set();
    const viewedCards = new Set();

    const $ = (selector) => document.querySelector(selector);
    const status = (message) => { $("#save-status").textContent = message; };

    function renderProgress() {
      $("#timeline-stat").textContent = `${localState.timeline_viewed} / 8`;
      $("#cards-stat").textContent = `${localState.cards_viewed} / 4`;
      $("#score-stat").textContent = `${localState.quiz_score} / 8`;
      const total = localState.timeline_viewed + localState.cards_viewed + localState.quiz_score;
      $("#progress-fill").style.width = `${Math.min(100, (total / 20) * 100)}%`;
      $("#progress-summary").textContent = `${total} von 20 Lernschritten aktiv bearbeitet`;
    }

    function updateQuizView() {
      document.querySelectorAll(".quiz-question").forEach(q => q.classList.add("hidden-question"));
      $("#quiz-complete").classList.add("hidden-question");
      if (localState.quiz_completed) {
        $("#quiz-progress").textContent = "Quiz abgeschlossen";
        $("#quiz-complete").classList.remove("hidden-question");
        $("#quiz-complete-score").textContent = `Dein Ergebnis: ${localState.quiz_score} von 8 Punkten`;
        $("#next-question").classList.add("hidden");
        return;
      }
      const question = document.querySelector(`[data-question="${currentQuestion}"]`);
      if (question) question.classList.remove("hidden-question");
      $("#quiz-progress").textContent = `Frage ${currentQuestion} von 8`;
    }

    function applyRecord(record) {
      localState = {
        quiz_score: Number(record.quiz_score || 0),
        quiz_completed: Boolean(record.quiz_completed),
        cards_viewed: Number(record.cards_viewed || 0),
        timeline_viewed: Number(record.timeline_viewed || 0)
      };
      renderProgress();
      updateQuizView();
    }

    const handler = {
      onDataChanged(data) {
        backendRecords = data;
        if (data.length) applyRecord(data[0]);
        else {
          renderProgress();
          updateQuizView();
          status("Dein Lernstand wird ab der ersten Aktion gespeichert.");
        }
      }
    };

    async function saveState(control) {
      if (control) {
        control.disabled = true;
        control.setAttribute("aria-busy", "true");
      }
      status("Lernstand wird gespeichert …");
      const record = { ...localState, updated_at: new Date().toISOString() };
      try {
        let result;
        if (backendRecords[0]) {
          result = await window.dataSdk.update({ ...backendRecords[0], ...record });
        } else {
          if (backendRecords.length >= 999) {
            status("Der Speicher ist voll. Der Lernstand konnte nicht gesichert werden.");
            return;
          }
          result = await window.dataSdk.create(record);
        }
        status(result.isOk ? "Lernstand gespeichert." : "Speichern konnte nicht bestätigt werden. Bitte erneut versuchen.");
      } catch (error) {
        status("Der Lernstand konnte gerade nicht gespeichert werden.");
      } finally {
        if (control) {
          control.disabled = false;
          control.removeAttribute("aria-busy");
        }
      }
    }

    async function initializeData() {
      status("Gespeicherten Lernstand laden …");
      const result = await window.dataSdk.init(handler);
      if (!result.isOk) status("Der gespeicherte Lernstand konnte nicht geladen werden. Du kannst trotzdem lernen.");
    }

    document.querySelectorAll(".timeline-button").forEach(button => {
      button.addEventListener("click", async () => {
        const year = button.dataset.year;
        document.querySelectorAll(".timeline-button").forEach(item => item.classList.remove("is-active"));
        document.querySelectorAll(".timeline-detail").forEach(item => item.classList.add("hidden"));
        button.classList.add("is-active");
        document.querySelector(`[data-detail="${year}"]`).classList.remove("hidden");
        if (!viewedYears.has(year)) {
          viewedYears.add(year);
          button.classList.add("is-viewed");
          localState.timeline_viewed = Math.min(8, localState.timeline_viewed + 1);
          renderProgress();
          await saveState(button);
        }
      });
    });

    document.querySelectorAll(".flip-card").forEach(card => {
      card.addEventListener("click", async () => {
        const flipped = card.classList.toggle("is-flipped");
        card.setAttribute("aria-pressed", String(flipped));
        const id = card.dataset.card;
        if (flipped && !viewedCards.has(id)) {
          viewedCards.add(id);
          localState.cards_viewed = Math.min(4, localState.cards_viewed + 1);
          renderProgress();
          await saveState(card);
        }
      });
    });

    document.querySelectorAll(".quiz-option").forEach(option => {
      option.addEventListener("click", async (event) => {
        const question = option.closest(".quiz-question");
        if (question.dataset.answered === "true") return;
        question.dataset.answered = "true";
        const correct = option.dataset.correct === "true";
        question.querySelectorAll(".quiz-option").forEach(button => {
          button.disabled = true;
          if (button.dataset.correct === "true") button.classList.add("correct");
        });
        if (correct) {
          localState.quiz_score = Math.min(8, localState.quiz_score + 1);
          option.closest(".quiz-question").querySelector(".quiz-feedback").textContent = "Richtig! Sehr gut.";
          option.closest(".quiz-question").querySelector(".quiz-feedback").className = "quiz-feedback feedback-good mt-5 min-h-6 font-semibold";
        } else {
          option.classList.add("incorrect");
          option.closest(".quiz-question").querySelector(".quiz-feedback").textContent = "Noch nicht richtig. Die grün markierte Antwort ist korrekt.";
          option.closest(".quiz-question").querySelector(".quiz-feedback").className = "quiz-feedback feedback-bad mt-5 min-h-6 font-semibold";
        }
        renderProgress();
        $("#next-question").classList.remove("hidden");
        await saveState(event.currentTarget);
      });
    });

    $("#next-question").addEventListener("click", async (event) => {
      if (currentQuestion < 8) {
        currentQuestion += 1;
        $("#next-question").classList.add("hidden");
        updateQuizView();
      } else {
        localState.quiz_completed = true;
        renderProgress();
        updateQuizView();
        await saveState(event.currentTarget);
      }
    });

    $("#reset-button").addEventListener("click", async (event) => {
      localState = { quiz_score: 0, quiz_completed: false, cards_viewed: 0, timeline_viewed: 0 };
      currentQuestion = 1;
      viewedYears.clear();
      viewedCards.clear();
      document.querySelectorAll(".timeline-button").forEach(item => item.classList.remove("is-active", "is-viewed"));
      document.querySelectorAll(".timeline-detail").forEach((item, index) => item.classList.toggle("hidden", index !== 0));
      document.querySelectorAll(".flip-card").forEach(card => { card.classList.remove("is-flipped"); card.setAttribute("aria-pressed", "false"); });
      document.querySelectorAll(".quiz-question").forEach(question => {
        question.dataset.answered = "false";
        question.querySelectorAll(".quiz-option").forEach(button => { button.disabled = false; button.classList.remove("correct", "incorrect"); });
        const feedback = question.querySelector(".quiz-feedback");
        feedback.textContent = "";
        feedback.className = "quiz-feedback mt-5 min-h-6 font-semibold";
      });
      $("#next-question").classList.add("hidden");
      renderProgress();
      updateQuizView();
      await saveState(event.currentTarget);
      status("Der Lernfortschritt wurde zurückgesetzt und gespeichert.");
    });

    document.addEventListener("DOMContentLoaded", () => {
      lucide.createIcons();
      renderProgress();
      updateQuizView();
      initializeData();
    });
  </script>
 
</body></html>
